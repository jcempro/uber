import {describe,it,expect} from 'vitest';
import type {ExactRational,ExactLimits} from './exact';

const limits:ExactLimits={maxInputLength:256,maxDigits:64};
const toRational=(n:string,d='1'):ExactRational=>({numerator:n,denominator:d});

  currency:'BRL',
  sourceBasis:'budget-calendar-period',
  amount:amount,
  sourcePeriod:sourcePeriod,
  calendar:{kind:'budget',...calendar}
});
const expectReadySource=(r:BudgetFinancialRatesResult,amount:ExactRational)=>{expect(r.status).toBe('conditional');expect(r.currency).toBe('BRL');expect(r.sourceBasis).toBe('budget-calendar-period');expect(r.source.status).toBe('ready');expect(r.source.sourcePeriod).toBe('month');expect(r.source.targetPeriod).toBe('month');expect(r.source.basis).toBe('budget-calendar-period');expect(r.sourceAmountMonthly).toBeUndefined();expect(r.sourceAmountMonthly).toBeUndefined();expect(r.source.value.numerator).toBe(amount.numerator);expect(r.source.value.denominator).toBe(amount.denominator);};
const expectUnavailable=(r:BudgetFinancialRatesResult,code:'invalid-budget'|'unsupported-input'|'source-unavailable',reasonIncludes:string)=>{expect(r.status).toBe('unavailable');expect(r.code).toBe(code);expect(r.reason).toContain(reasonIncludes);expect(r.source).toBeNull();};

const expectWorkingRateUnavailable=(r:{status:'ready'|'unavailable',unit?:string,amountPeriod?:string,value?:ExactRational,denominator?:ExactRational,code?:string,reason?:string,diagnostic?:unknown},code: string, reasonIncludes: string)=>{expect(r.status).toBe('unavailable');expect(r.code).toBe(code);expect(r.reason).toContain(reasonIncludes);};
